# Chapter 14, solution 2

```
function simboot(k::Integer, n::Integer)
    result = [boot(n) for _ in 1:k]
    mv = mean(result)
    sdv = std(result)
    lo95 = mv - 1.96 * sdv / sqrt(k)
    hi95 = mv + 1.96 * sdv / sqrt(k)
    return (; k, n, mv, lo95, hi95)
end
```
